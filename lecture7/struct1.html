Rust Structs and Enums Examples

Note: These are Rust code examples. To run them, you'll need the Rust compiler installed.

Basic Examples

Simple Struct Example

// Define a simple struct
struct Book {
    title: String,
    author: String,
    pages: u32,
}

fn main() {
    // Create an instance of the Book struct
    let book = Book {
        title: String::from("The Rust Programming Language"),
        author: String::from("Steve Klabnik and Carol Nichols"),
        pages: 519,
    };

    println!("Book: {} by {} ({} pages)", 
             book.title, book.author, book.pages);
}

Simple Enum Example

// Define a simple enum
enum Direction {
    North,
    South,
    East,
    West,
}

fn main() {
    // Use the enum
    let my_direction = Direction::North;

    // Match against enum variants
    match my_direction {
        Direction::North => println!("Heading north!"),
        Direction::South => println!("Heading south!"),
        Direction::East => println!("Heading east!"),
        Direction::West => println!("Heading west!"),
    }
}

Nested Structures

Struct Containing Another Struct

// Define address struct
struct Address {
    street: String,
    city: String,
    zip_code: String,
}

// Define person struct containing address
struct Person {
    name: String,
    age: u8,
    address: Address,  // Struct within struct
}

fn main() {
    let person = Person {
        name: String::from("Alice"),
        age: 30,
        address: Address {
            street: String::from("123 Main St"),
            city: String::from("Techville"),
            zip_code: String::from("12345"),
        },
    };

    println!("{} lives at {}, {}, {}",
        person.name,
        person.address.street,
        person.address.city,
        person.address.zip_code);
}

Enum Containing Structs

// Define shape structs
struct Circle {
    radius: f64,
}

struct Rectangle {
    width: f64,
    height: f64,
}

// Enum containing different shape structs
enum Shape {
    Circle(Circle),
    Rectangle(Rectangle),
}

impl Shape {
    fn area(&self) -> f64 {
        match self {
            Shape::Circle(c) => std::f64::consts::PI * c.radius * c.radius,
            Shape::Rectangle(r) => r.width * r.height,
        }
    }
}

fn main() {
    let circle = Shape::Circle(Circle { radius: 5.0 });
    let rectangle = Shape::Rectangle(Rectangle { width: 4.0, height: 6.0 });

    println!("Circle area: {:.2}", circle.area());
    println!("Rectangle area: {:.2}", rectangle.area());
}

Struct Containing Enum

// Define account status enum
enum AccountStatus {
    Active,
    Inactive,
    Suspended,
    Banned,
}

// Define user struct containing enum
struct User {
    username: String,
    email: String,
    status: AccountStatus,
}

fn print_user_status(user: &User) {
    match user.status {
        AccountStatus::Active => println!("{} is active", user.username),
        AccountStatus::Inactive => println!("{} is inactive", user.username),
        AccountStatus::Suspended => println!("{} is suspended", user.username),
        AccountStatus::Banned => println!("{} is banned", user.username),
    }
}

fn main() {
    let user1 = User {
        username: String::from("alice123"),
        email: String::from("alice@example.com"),
        status: AccountStatus::Active,
    };

    let user2 = User {
        username: String::from("bob456"),
        email: String::from("bob@example.com"),
        status: AccountStatus::Suspended,
    };

    print_user_status(&user1);
    print_user_status(&user2);
}

Key Points to Remember

  • Structs can contain other structs as fields
  • Enums can contain structs as variant data
  • Structs can contain enums as fields
  • These combinations allow for modeling complex data relationships
  • Match expressions work well with nested enums and structs
